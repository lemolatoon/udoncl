using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using Lemolatoon.Udoncl.Editor;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using UnityEngine.TestTools;
using VRC.Udon;

namespace Lemolatoon.Udoncl.Editor.Tests
{
    public sealed class PublicTinyLlamaPrefillLifecycleTests
    {
        private const string GeneratedRoot = "Assets/udoncl-generated/tinyllama_real_prefill";
        private const string BindingClass = "UdonclTinyLlamaPrefillBinding";
        private const string ControllerClass = "UdonclTinyLlamaPrefillTestController";
        private const string GeneratedNamespace = "Lemolatoon.Udoncl.Generated.";
        private const string ControllerSource =
            "Assets/UdonclSampleValidation/" + ControllerClass + ".cs";
        private const string ControllerProgram =
            GeneratedRoot + "/Tests/" + ControllerClass + ".asset";
        private ValidatedOnnxOracle oracle;

        [OneTimeSetUp]
        public void ImportPrefillArtifact()
        {
            string artifactDirectory = Environment.GetEnvironmentVariable(
                "UDONCL_PUBLIC_TINYLLAMA_PREFILL_UNITY_BUNDLE"
            );
            Assert.IsFalse(
                string.IsNullOrWhiteSpace(artifactDirectory),
                "The real TinyLlama prefill artifact path is required."
            );
            artifactDirectory = Path.GetFullPath(artifactDirectory);
            ValidatedArtifact artifact = ArtifactManifestValidator.Validate(artifactDirectory);
            Assert.AreEqual("tinyllama_real_prefill", artifact.ProgramName);
            if (!AssetDatabase.IsValidFolder(GeneratedRoot))
                Assert.AreEqual(GeneratedRoot, UdonclArtifactImporter.Import(artifactDirectory));
            CollectionAssert.AreEqual(
                File.ReadAllBytes(Path.Combine(artifactDirectory, "binding.cs")),
                File.ReadAllBytes(GeneratedRoot + "/Bindings/" + BindingClass + ".cs")
            );
            Assert.NotNull(AssetDatabase.LoadAssetAtPath<MonoScript>(ControllerSource));
            if (!AssetDatabase.IsValidFolder(GeneratedRoot + "/Tests"))
                AssetDatabase.CreateFolder(GeneratedRoot, "Tests");
            UdonclArtifactImporter.EnsureCompiledUdonSharpProgramAsset(
                GeneratedRoot + "/Bindings/" + BindingClass + ".cs",
                GeneratedRoot + "/Bindings/" + BindingClass + ".asset"
            );
            UdonclArtifactImporter.EnsureCompiledUdonSharpProgramAsset(
                ControllerSource,
                ControllerProgram
            );
            GeneratedBindingFinalizer.FinalizeAvailableBindings();

            string oracleDirectory = Environment.GetEnvironmentVariable(
                "UDONCL_PUBLIC_TINYLLAMA_PREFILL_ORACLE"
            );
            Assert.IsFalse(
                string.IsNullOrWhiteSpace(oracleDirectory),
                "The TinyLlama prefill oracle directory is required."
            );
            oracle = OnnxOracleSidecar.Load(oracleDirectory);
            Assert.GreaterOrEqual(oracle.Inputs.Count, 3);
            CollectionAssert.AreEqual(
                new[] { "input_ids", "attention_mask", "position_ids" },
                new[]
                {
                    oracle.Inputs[0].Name,
                    oracle.Inputs[1].Name,
                    oracle.Inputs[2].Name,
                }
            );
            for (int index = 0; index < 3; ++index)
            {
                Assert.AreEqual("int64", oracle.Inputs[index].DataType);
                Assert.Greater(oracle.Inputs[index].Values.Length, 0);
            }
            for (int index = 3; index < oracle.Inputs.Count; ++index)
            {
                ValidatedOnnxOracleInput input = oracle.Inputs[index];
                StringAssert.StartsWith("past_key_values.", input.Name);
                Assert.AreEqual("float32", input.DataType);
                Assert.AreEqual(0, input.Values.Length);
                CollectionAssert.Contains(input.Shape, 0L);
            }
            Assert.AreEqual(45, oracle.Outputs.Count);
            Assert.AreEqual(128000, oracle.Outputs[0].ElementCount);
            for (int index = 1; index < oracle.Outputs.Count; ++index)
                Assert.AreEqual(1024, oracle.Outputs[index].ElementCount);
            // The controller reads logits and then the presents in ONNX
            // order, each through the method named after its Host variable.
            // Each output's public buffer is found by name; distinct names
            // over equal counts make the pairing one to one.
            int publicOutputs = 0;
            foreach (ValidatedDeviceBuffer buffer in artifact.DeviceBuffers)
                if (buffer.PublicAccess == "output")
                    ++publicOutputs;
            Assert.AreEqual(oracle.Outputs.Count, publicOutputs);
            HashSet<string> pairedNames = new HashSet<string>();
            for (int index = 0; index < oracle.Outputs.Count; ++index)
            {
                ValidatedOnnxOracleOutput output = oracle.Outputs[index];
                Assert.AreEqual(index, output.Index, output.Name + " is out of order.");
                int present = index - 1;
                Assert.AreEqual(
                    index == 0
                        ? "logits"
                        : "present." + present / 2 + (present % 2 == 0 ? ".key" : ".value"),
                    output.Name
                );
                ValidatedDeviceBuffer buffer = OnnxPublicBuffers.Find(
                    artifact.DeviceBuffers,
                    "output",
                    output.Index,
                    output.Name
                );
                Assert.IsTrue(
                    pairedNames.Add(buffer.Name),
                    output.Name + " shares a public buffer."
                );
                Assert.AreEqual("float32", buffer.ElementType);
                Assert.AreEqual(output.ElementCount * sizeof(float), buffer.LogicalBytes);
            }
        }

        [UnityTest]
        public IEnumerator GeneratedUSharpBindingMatchesAllRealPrefillOutputs()
        {
            yield return new EnterPlayMode();

            GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(
                GeneratedRoot + "/Prefabs/" + BindingClass + ".prefab"
            );
            Assert.NotNull(prefab);
            GameObject instance = UnityEngine.Object.Instantiate(prefab);
            UdonBehaviour controller = UdonclArtifactImporter.AddCompiledUdonSharpBehaviour(
                instance,
                ControllerProgram
            );
            UdonclArtifactImporter.InitializeUdonBehaviours(instance);
            yield return null;

            UdonBehaviour binding = UdonclArtifactImporter.FindUdonSharpBacking(
                instance,
                GeneratedNamespace + BindingClass
            );
            Assert.NotNull(binding);
            Assert.NotNull(controller);
            float[] logits = new float[oracle.Outputs[0].ElementCount];
            float[] presentScratch = new float[oracle.Outputs[1].ElementCount];
            float[] presentKv = new float[(oracle.Outputs.Count - 1) * presentScratch.Length];
            UdonclArtifactImporter.SetUdonProgramVariable(controller, "binding", binding);
            UdonclArtifactImporter.SetUdonProgramVariable(
                controller,
                "inputIds",
                oracle.Inputs[0].ReadInt64Values()
            );
            UdonclArtifactImporter.SetUdonProgramVariable(
                controller,
                "attentionMask",
                oracle.Inputs[1].ReadInt64Values()
            );
            UdonclArtifactImporter.SetUdonProgramVariable(
                controller,
                "positionIds",
                oracle.Inputs[2].ReadInt64Values()
            );
            UdonclArtifactImporter.SetUdonProgramVariable(controller, "logits", logits);
            UdonclArtifactImporter.SetUdonProgramVariable(
                controller,
                "presentScratch",
                presentScratch
            );
            UdonclArtifactImporter.SetUdonProgramVariable(controller, "presentKv", presentKv);

            // Controller phases: 1 initialize, 2 run, 3 logits readback in
            // flight, 4 + n after the n-th readback callback, 49 done. Each
            // readback is requested from the previous completion callback,
            // so the per-frame poll sees every request -> callback interval.
            UdonclFixtureTiming timing = new UdonclFixtureTiming("tinyllama_prefill");
            const int timeoutFrames = 36000;
            int frame = 0;
            bool completed = false;
            try
            {
                timing.Begin("initialize", 0);
                UdonclArtifactImporter.SendUdonCustomEvent(controller, "Run");
                while (frame < timeoutFrames && !completed)
                {
                    ++frame;
                    yield return null;
                    timing.Frame();
                    completed = (bool)
                        UdonclArtifactImporter.GetUdonProgramVariable(controller, "completed");
                    int currentPhase = (int)
                        UdonclArtifactImporter.GetUdonProgramVariable(controller, "phase");
                    int presentIndex = (int)
                        UdonclArtifactImporter.GetUdonProgramVariable(controller, "presentIndex");
                    int readbacks = currentPhase >= 4 ? 1 + presentIndex : 0;
                    timing.Readbacks(
                        readbacks,
                        currentPhase >= 3 && !completed && readbacks < oracle.Outputs.Count,
                        readbacks < oracle.Outputs.Count ? oracle.Outputs[readbacks].Name : null
                    );
                    if (currentPhase >= 49)
                        timing.Phase("done", 0);
                    else if (currentPhase >= 3)
                        timing.Phase("read-outputs", 0);
                    else if (currentPhase == 2)
                        timing.Phase("run", 1);
                }
            }
            finally
            {
                // Restores the frame-rate settings and logs the totals.
                timing.End();
            }
            int phase = (int)UdonclArtifactImporter.GetUdonProgramVariable(controller, "phase");
            int status = (int)UdonclArtifactImporter.GetUdonProgramVariable(controller, "status");
            int libraryStatus = (int)
                UdonclArtifactImporter.GetUdonProgramVariable(controller, "libraryStatus");
            float[] actualLogits = (float[])
                UdonclArtifactImporter.GetUdonProgramVariable(controller, "logits");
            float[] actualPresent = (float[])
                UdonclArtifactImporter.GetUdonProgramVariable(controller, "presentKv");
            UnityEngine.Object.Destroy(instance);
            yield return new ExitPlayMode();

            Assert.IsTrue(completed, "TinyLlama prefill timed out at phase " + phase + ".");
            Assert.AreEqual(0, status, "Generated binding failed at phase " + phase + ".");
            Assert.AreEqual(0, libraryStatus, "C Host failed at phase " + phase + ".");
            oracle.RequireFloat32Close(0, actualLogits, 0.001f, 0.001f);
            for (int index = 1; index < oracle.Outputs.Count; ++index)
                oracle.RequireFloat32Close(
                    index,
                    actualPresent,
                    (index - 1) * presentScratch.Length,
                    0.001f,
                    0.001f
                );
            string result = string.Format(
                "tinyllama_prefill frames={0} outputs={1}{2} run_frames={3} run_ms={4}",
                frame,
                oracle.Outputs.Count,
                timing.Summary,
                timing.PhaseFrames("run"),
                timing.PhaseMillisecondsText("run")
            );
            TestContext.Progress.WriteLine(result);
            // NUnit progress output does not reach the Editor log.
            Debug.Log("udoncl " + result);
        }
    }
}
