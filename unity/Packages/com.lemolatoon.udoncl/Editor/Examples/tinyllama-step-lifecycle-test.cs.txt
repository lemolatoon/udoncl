using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Text;
using Lemolatoon.Udoncl.Editor;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using UnityEngine.TestTools;
using VRC.Udon;

namespace Lemolatoon.Udoncl.Editor.Tests
{
    // Step-level numerical gate for the M=1 fixed-capacity graph. Every ORT step
    // sidecar (all inputs including the rolling past K/V, all outputs) is fed
    // through the generated public-buffer binding of the step validation
    // artifact, and every output is compared. The report names each output
    // with its maximum error, so a failing run localizes the first diverging
    // tensor instead of only a wrong token.
    //
    // The step graph is plain ONNX: its past and present K/V are ordinary
    // inputs and outputs, each bound to the UDONCL_PUBLIC buffer of the same
    // name, so every present is read straight from its output buffer. The
    // controller writes and reads them through the binding's named methods,
    // in ONNX order; this test hands it the sidecar tensors in that order.
    public sealed class PublicTinyLlamaStepValidationTests
    {
        private const string GeneratedRoot = "Assets/udoncl-generated/tinyllama_real_m1_step";
        private const string BindingClass = "UdonclTinyLlamaStepBinding";
        private const string ControllerClass = "UdonclTinyLlamaStepTestController";
        private const string GeneratedNamespace = "Lemolatoon.Udoncl.Generated.";
        private const string ControllerSource =
            "Assets/UdonclSampleValidation/" + ControllerClass + ".cs";
        private const string ControllerProgram =
            GeneratedRoot + "/Tests/" + ControllerClass + ".asset";
        // Mirrors UdonclTinyLlamaStepTestController's phases.
        private const int PhaseRun = 3;
        private const int PhaseReadOutputs = 4;
        private const int PhaseDone = 6;
        private static readonly string[] PhaseNames =
        {
            "not-started",
            "initialize",
            "write-inputs",
            "run",
            "read-outputs",
            "shutdown",
            "done",
        };
        // Mirrors UdonclTinyLlamaStepTestController's tensor counts.
        private const int InputCount = 47;
        private const int OutputCount = 45;
        private const float AbsoluteTolerance = 0.001f;
        private const float RelativeTolerance = 0.001f;

        [OneTimeSetUp]
        public void ImportStepArtifact()
        {
            string artifactDirectory = RequireDirectory(
                "UDONCL_PUBLIC_TINYLLAMA_STEP_UNITY_BUNDLE",
                "The M=1 step validation artifact path is required."
            );
            ValidatedArtifact artifact = ArtifactManifestValidator.Validate(artifactDirectory);
            Assert.AreEqual("tinyllama_real_m1_step", artifact.ProgramName);
            if (!AssetDatabase.IsValidFolder(GeneratedRoot))
                Assert.AreEqual(GeneratedRoot, UdonclArtifactImporter.Import(artifactDirectory));
            CollectionAssert.AreEqual(
                File.ReadAllBytes(Path.Combine(artifactDirectory, "binding.cs")),
                File.ReadAllBytes(GeneratedRoot + "/Bindings/" + BindingClass + ".cs")
            );
            Assert.NotNull(
                AssetDatabase.LoadAssetAtPath<MonoScript>(ControllerSource),
                "The step test controller was not installed."
            );
            if (!AssetDatabase.IsValidFolder(GeneratedRoot + "/Tests"))
                AssetDatabase.CreateFolder(GeneratedRoot, "Tests");
            UdonclArtifactImporter.EnsureCompiledUdonSharpProgramAsset(
                GeneratedRoot + "/Bindings/" + BindingClass + ".cs",
                GeneratedRoot + "/Bindings/" + BindingClass + ".asset"
            );
            UdonclArtifactImporter.EnsureCompiledUdonSharpProgramAsset(
                ControllerSource,
                ControllerProgram
            );
            GeneratedBindingFinalizer.FinalizeAvailableBindings();
        }

        // EnterPlayMode reloads the domain and the Unity Test Framework then
        // restores only this iterator's state number: everything is loaded
        // after that yield, and the method uses no closures.
        [UnityTest]
        public IEnumerator GeneratedUSharpBindingMatchesM1StepOutputs()
        {
            yield return new EnterPlayMode();

            string artifactDirectory = RequireDirectory(
                "UDONCL_PUBLIC_TINYLLAMA_STEP_UNITY_BUNDLE",
                "The M=1 step validation artifact path is required."
            );
            string oracleRoot = RequireDirectory(
                "UDONCL_PUBLIC_TINYLLAMA_STEP_ORACLES",
                "The M=1 step oracle directory (containing step-* sidecars) is required."
            );
            ValidatedArtifact artifact = ArtifactManifestValidator.Validate(artifactDirectory);
            List<ValidatedDeviceBuffer> inputBuffers = new List<ValidatedDeviceBuffer>();
            List<ValidatedDeviceBuffer> outputBuffers = new List<ValidatedDeviceBuffer>();
            foreach (ValidatedDeviceBuffer buffer in artifact.DeviceBuffers)
            {
                if (buffer.PublicAccess == "input")
                    inputBuffers.Add(buffer);
                else if (buffer.PublicAccess == "output")
                    outputBuffers.Add(buffer);
            }
            string[] stepDirectories = Directory.GetDirectories(oracleRoot, "step-*");
            Array.Sort(stepDirectories, StringComparer.Ordinal);
            Assert.Greater(stepDirectories.Length, 0, "No step-* sidecars in " + oracleRoot);

            StringBuilder report = new StringBuilder();
            int failures = 0;
            for (int step = 0; step < stepDirectories.Length; ++step)
            {
                string stepName = Path.GetFileName(stepDirectories[step]);
                ValidatedOnnxOracle oracle = OnnxOracleSidecar.Load(stepDirectories[step]);
                Assert.AreEqual(
                    InputCount,
                    oracle.Inputs.Count,
                    stepName + ": the sidecar input count differs from the step graph."
                );
                Assert.AreEqual(
                    OutputCount,
                    oracle.Outputs.Count,
                    stepName + ": the sidecar output count differs from the step graph."
                );
                Assert.AreEqual(
                    inputBuffers.Count,
                    oracle.Inputs.Count,
                    stepName + ": public input buffer count differs from the sidecar."
                );
                Assert.AreEqual(
                    outputBuffers.Count,
                    oracle.Outputs.Count,
                    stepName + ": public output buffer count differs from the sidecar."
                );
                // The controller addresses each tensor by its ONNX index, so
                // the sidecar must list them in that order. Each tensor's
                // public buffer (the Host variable bound to it) is found by
                // name; distinct names over equal counts make the pairing one
                // to one.
                HashSet<string> pairedNames = new HashSet<string>();
                object[] inputValues = new object[inputBuffers.Count];
                for (int index = 0; index < inputBuffers.Count; ++index)
                {
                    ValidatedOnnxOracleInput input = oracle.Inputs[index];
                    Assert.AreEqual(
                        StepInputName(index),
                        input.Name,
                        stepName + ": sidecar input " + index + " is not the ONNX input."
                    );
                    ValidatedDeviceBuffer buffer = OnnxPublicBuffers.Find(
                        inputBuffers,
                        "input",
                        index,
                        input.Name
                    );
                    Assert.IsTrue(
                        pairedNames.Add(buffer.Name),
                        stepName + ": input " + input.Name + " shares a public buffer."
                    );
                    Assert.AreEqual(
                        input.DataType,
                        buffer.ElementType,
                        stepName + ": input " + input.Name + " element type differs."
                    );
                    if (input.DataType == "int64")
                    {
                        inputValues[index] = input.ReadInt64Values();
                        Assert.AreEqual(
                            input.Values.Length * sizeof(long),
                            buffer.LogicalBytes,
                            stepName + ": input " + input.Name + " size differs."
                        );
                    }
                    else
                    {
                        Assert.AreEqual(
                            "float32",
                            input.DataType,
                            stepName + ": input " + input.Name + " type."
                        );
                        inputValues[index] = ToFloats(input.Values);
                        Assert.AreEqual(
                            input.Values.Length * sizeof(float),
                            buffer.LogicalBytes,
                            stepName + ": input " + input.Name + " size differs."
                        );
                    }
                }
                object[] outputValues = new object[outputBuffers.Count];
                for (int index = 0; index < outputBuffers.Count; ++index)
                {
                    ValidatedOnnxOracleOutput output = oracle.Outputs[index];
                    Assert.AreEqual(
                        index,
                        output.Index,
                        stepName + ": sidecar output " + output.Name + " is out of order."
                    );
                    Assert.AreEqual(
                        StepOutputName(index),
                        output.Name,
                        stepName + ": sidecar output " + index + " is not the ONNX output."
                    );
                    ValidatedDeviceBuffer buffer = OnnxPublicBuffers.Find(
                        outputBuffers,
                        "output",
                        output.Index,
                        output.Name
                    );
                    Assert.IsTrue(
                        pairedNames.Add(buffer.Name),
                        stepName + ": output " + output.Name + " shares a public buffer."
                    );
                    Assert.AreEqual(
                        "float32",
                        buffer.ElementType,
                        stepName + ": output " + output.Name + " element type differs."
                    );
                    Assert.AreEqual(
                        output.ElementCount * sizeof(float),
                        buffer.LogicalBytes,
                        stepName + ": output " + output.Name + " size differs."
                    );
                    outputValues[index] = new float[oracle.Outputs[index].ElementCount];
                }

                GameObject prefab = AssetDatabase.LoadAssetAtPath<GameObject>(
                    GeneratedRoot + "/Prefabs/" + BindingClass + ".prefab"
                );
                Assert.NotNull(prefab, "The step binding prefab is missing.");
                GameObject instance = UnityEngine.Object.Instantiate(prefab);
                UdonBehaviour controller = UdonclArtifactImporter.AddCompiledUdonSharpBehaviour(
                    instance,
                    ControllerProgram
                );
                UdonclArtifactImporter.InitializeUdonBehaviours(instance);
                yield return null;
                UdonBehaviour binding = UdonclArtifactImporter.FindUdonSharpBacking(
                    instance,
                    GeneratedNamespace + BindingClass
                );
                Assert.NotNull(binding, "The step binding UdonBehaviour was not found.");
                Assert.NotNull(controller, "The step test controller was not added.");
                UdonclArtifactImporter.SetUdonProgramVariable(controller, "binding", binding);
                UdonclArtifactImporter.SetUdonProgramVariable(controller, "inputValues", inputValues);
                UdonclArtifactImporter.SetUdonProgramVariable(
                    controller,
                    "outputValues",
                    outputValues
                );
                Assert.NotNull(
                    UdonclArtifactImporter.GetUdonProgramVariable(controller, "outputValues"),
                    "The step test controller program is not initialized."
                );

                // The run phase is exactly one M=1 model step. Each output
                // readback is requested from the previous completion
                // callback, so the per-frame poll sees every request ->
                // callback interval.
                UdonclFixtureTiming timing = new UdonclFixtureTiming(
                    "tinyllama_m1_step",
                    "step=" + stepName
                );
                const int timeoutFrames = 36000;
                int frame = 0;
                bool completed = false;
                try
                {
                    timing.Begin(PhaseName(1), 0);
                    UdonclArtifactImporter.SendUdonCustomEvent(controller, "Run");
                    while (frame < timeoutFrames && !completed)
                    {
                        ++frame;
                        yield return null;
                        timing.Frame();
                        completed = ReadBool(controller, "completed");
                        int currentPhase = ReadInt(controller, "phase");
                        int readbacks =
                            currentPhase >= PhaseReadOutputs
                                ? ReadInt(controller, "transferIndex")
                                : 0;
                        timing.Readbacks(
                            readbacks,
                            currentPhase == PhaseReadOutputs && !completed,
                            readbacks < oracle.Outputs.Count
                                ? oracle.Outputs[readbacks].Name
                                : null
                        );
                        timing.Phase(
                            PhaseName(currentPhase),
                            currentPhase == PhaseRun ? 1 : 0
                        );
                    }
                }
                finally
                {
                    // Restores the frame-rate settings and logs the totals.
                    timing.End();
                }
                int phase = ReadInt(controller, "phase");
                int status = ReadInt(controller, "status");
                int libraryStatus = ReadInt(controller, "libraryStatus");
                object[] actual = (object[])
                    UdonclArtifactImporter.GetUdonProgramVariable(controller, "outputValues");
                UnityEngine.Object.Destroy(instance);
                yield return null;

                string header =
                    stepName
                    + " phase="
                    + phase
                    + " status="
                    + status
                    + " libraryStatus="
                    + libraryStatus
                    + " frames="
                    + frame
                    + timing.Summary
                    + " run_frames="
                    + timing.PhaseFrames(PhaseName(PhaseRun))
                    + " run_ms="
                    + timing.PhaseMillisecondsText(PhaseName(PhaseRun));
                Debug.Log("udoncl tinyllama_m1_step " + header);
                report.Append(header).Append('\n');
                if (!completed || phase != PhaseDone || status != 0 || libraryStatus != 0)
                {
                    ++failures;
                    continue;
                }
                for (int index = 0; index < oracle.Outputs.Count; ++index)
                {
                    string line = CompareOutput(stepName, oracle, index, (float[])actual[index]);
                    if (line.IndexOf(" mismatches=0/", StringComparison.Ordinal) < 0)
                        ++failures;
                    Debug.Log("udoncl tinyllama_m1_step " + line);
                    report.Append(line).Append('\n');
                }
            }
            yield return new ExitPlayMode();
            Assert.AreEqual(0, failures, "M=1 step outputs differ from ORT:\n" + report);
        }

        private static string CompareOutput(
            string stepName,
            ValidatedOnnxOracle oracle,
            int index,
            float[] actual
        )
        {
            float[] expected = oracle.ReadFloat32Output(index);
            double maxError = 0.0;
            int mismatches = 0;
            int firstMismatch = -1;
            for (int element = 0; element < expected.Length; ++element)
            {
                float value = actual[element];
                double error = Math.Abs((double)value - expected[element]);
                float tolerance =
                    AbsoluteTolerance + RelativeTolerance * Math.Abs(expected[element]);
                if (float.IsNaN(value) || float.IsInfinity(value) || error > tolerance)
                {
                    if (firstMismatch < 0)
                        firstMismatch = element;
                    ++mismatches;
                }
                if (!double.IsNaN(error) && error > maxError)
                    maxError = error;
            }
            string line =
                stepName
                + " output="
                + oracle.Outputs[index].Name
                + " max_abs_error="
                + maxError.ToString("R")
                + " mismatches="
                + mismatches
                + "/"
                + expected.Length;
            if (firstMismatch >= 0)
                line +=
                    " first="
                    + firstMismatch
                    + " expected="
                    + expected[firstMismatch].ToString("R")
                    + " actual="
                    + actual[firstMismatch].ToString("R");
            if (index == 0)
                line += " argmax_expected=" + ArgMax(expected) + " argmax_actual=" + ArgMax(actual);
            return line;
        }

        private static int ArgMax(float[] values)
        {
            int best = 0;
            for (int index = 1; index < values.Length; ++index)
                if (values[index] > values[best])
                    best = index;
            return best;
        }

        // The ONNX name of step input `index`: input_ids, attention_mask,
        // position_ids, then the past key and value of each layer.
        private static string StepInputName(int index)
        {
            if (index == 0)
                return "input_ids";
            if (index == 1)
                return "attention_mask";
            if (index == 2)
                return "position_ids";
            int past = index - 3;
            return "past_key_values." + past / 2 + (past % 2 == 0 ? ".key" : ".value");
        }

        // The ONNX name of step output `index`: logits, then the present key
        // and value of each layer.
        private static string StepOutputName(int index)
        {
            if (index == 0)
                return "logits";
            int present = index - 1;
            return "present." + present / 2 + (present % 2 == 0 ? ".key" : ".value");
        }

        private static float[] ToFloats(double[] values)
        {
            float[] result = new float[values.Length];
            for (int index = 0; index < values.Length; ++index)
                result[index] = (float)values[index];
            return result;
        }

        private static string RequireDirectory(string variable, string message)
        {
            string path = Environment.GetEnvironmentVariable(variable);
            Assert.IsFalse(string.IsNullOrWhiteSpace(path), message + " (" + variable + ")");
            path = Path.GetFullPath(path);
            Assert.IsTrue(Directory.Exists(path), message + " Missing: " + path);
            return path;
        }

        private static string PhaseName(int phase)
        {
            return phase >= 0 && phase < PhaseNames.Length ? PhaseNames[phase] : "unknown";
        }

        private static object ReadVariable(UdonBehaviour controller, string symbol)
        {
            object value = UdonclArtifactImporter.GetUdonProgramVariable(controller, symbol);
            Assert.NotNull(value, "Step controller variable '" + symbol + "' is unset.");
            return value;
        }

        private static int ReadInt(UdonBehaviour controller, string symbol)
        {
            object value = ReadVariable(controller, symbol);
            Assert.IsInstanceOf<int>(value, "Step controller variable '" + symbol + "' is not int.");
            return (int)value;
        }

        private static bool ReadBool(UdonBehaviour controller, string symbol)
        {
            object value = ReadVariable(controller, symbol);
            Assert.IsInstanceOf<bool>(
                value,
                "Step controller variable '" + symbol + "' is not bool."
            );
            return (bool)value;
        }
    }
}
