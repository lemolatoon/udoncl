    // artifact. The Editor test supplies the value of every graph input and a
    // destination for every graph output, both in ONNX order; the controller
    // writes and reads them through the binding's methods named after the

        public const int PhaseInitialize = 1;
        public const int PhaseWriteInputs = 2;
        public const int PhaseRun = 3;
        public const int PhaseReadOutputs = 4;
        public const int PhaseShutdown = 5;
        public const int PhaseDone = 6;

        // input_ids, attention_mask, position_ids and the past key and value
        // of the 22 layers; logits and the present key and value of each.
        public const int InputCount = 47;
        public const int OutputCount = 45;

        public UdonclTinyLlamaStepBinding binding;
        public object[] inputValues;
        public object[] outputValues;
        public bool completed;
        public int status;
        public int libraryStatus;
        public int phase;
        public int transferIndex;

        public void Run()
        {
            completed = false;
            status = UdonclTinyLlamaStepBinding.StatusSuccess;
            libraryStatus = 0;
            transferIndex = 0;
            phase = PhaseInitialize;
            if (
                binding == null
                || inputValues == null
                || inputValues.Length != InputCount
                || outputValues == null
                || outputValues.Length != OutputCount
            )
            {
                status = UdonclTinyLlamaStepBinding.StatusInvalidArgument;
                completed = true;
                return;
            }
            // Initialize enqueues no Device work, so it completes at once.
            InitializeBinding();
        }

        private void InitializeBinding()
        {
            status = binding.TryInitialize(out libraryStatus);
            if (!Succeeded())
                return;
            phase = PhaseWriteInputs;
            transferIndex = 0;
            SendCustomEventDelayedFrames(nameof(_WriteNextInput), 1);
        }

        // One input per frame keeps each packing loop well inside the Udon
        // per-event VM time limit.
        public void _WriteNextInput()
        {
            if (completed)
                return;
            if (transferIndex >= InputCount)
            {
                phase = PhaseRun;
                status = binding.StartRun(this, nameof(OnRunComplete));
                if (status != UdonclTinyLlamaStepBinding.StatusSuccess)
                    completed = true;
                return;
            }
            status = WriteInput(transferIndex, inputValues[transferIndex]);
            if (status != UdonclTinyLlamaStepBinding.StatusSuccess)
            {
                completed = true;
                return;
            }
            ++transferIndex;
            SendCustomEventDelayedFrames(nameof(_WriteNextInput), 1);
        }

        public void OnRunComplete()
        {
            status = binding.GetRunStatus();
            if (status == UdonclTinyLlamaStepBinding.StatusSuccess)
                status = binding.TryGetRunResult(out libraryStatus);
            if (!Succeeded())
                return;
            phase = PhaseReadOutputs;
            transferIndex = 0;
            StartNextOutput();
        }

        private void StartNextOutput()
        {
            status = StartReadOutput(transferIndex, (float[])outputValues[transferIndex]);
            if (status != UdonclTinyLlamaStepBinding.StatusSuccess)

        public void OnOutputComplete()
        {
            status = binding.GetLastReadbackStatus();
            if (status != UdonclTinyLlamaStepBinding.StatusSuccess)
            {
                completed = true;
                return;
            }
            ++transferIndex;
            if (transferIndex < OutputCount)
            {
                StartNextOutput();
                return;
            }
            phase = PhaseShutdown;
            status = binding.TryShutdown(out libraryStatus);
            if (!Succeeded())
                return;
            phase = PhaseDone;
            completed = true;
        }

        private bool Succeeded()
        {
            if (status == UdonclTinyLlamaStepBinding.StatusSuccess && libraryStatus == 0)
                return true;
            completed = true;
            return false;
        }
