cmake_minimum_required(VERSION 3.30...4.2)
project(udoncl_opencl_host_example LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/Runtime/OpenCL" "${CMAKE_CURRENT_BINARY_DIR}/opencl-runtime")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/Testing/NativeFragment" "${CMAKE_CURRENT_BINARY_DIR}/fragment-runtime")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/Testing/NativeOpenCL" "${CMAKE_CURRENT_BINARY_DIR}/native-opencl-runtime")
add_executable(opencl-host main.cpp "${UDONCL_HOST_OBJECT}" "${UDONCL_ORACLE_OBJECT}")
set_source_files_properties("${UDONCL_HOST_OBJECT}" "${UDONCL_ORACLE_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
target_include_directories(opencl-host PRIVATE "${UDONCL_BINDINGS_DIR}")
target_compile_features(opencl-host PRIVATE cxx_std_17)
set_target_properties(opencl-host PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(opencl-host PRIVATE VRCNativeOpenCLRuntime)

option(UDONCL_BUILD_RUNTIME_TESTS "Include OpenCL contract checks" OFF)
if(UDONCL_BUILD_RUNTIME_TESTS)
  target_sources(opencl-host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/runtime/opencl.c")
  target_compile_definitions(opencl-host PRIVATE UDONCL_BUILD_RUNTIME_TESTS=1)
endif()

if(DEFINED UDONCL_ORIGINAL_HOST_OBJECT)
  add_executable(opencl-host-original main.cpp "${UDONCL_ORIGINAL_HOST_OBJECT}" "${UDONCL_ORACLE_OBJECT}")
  set_source_files_properties("${UDONCL_ORIGINAL_HOST_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  target_compile_features(opencl-host-original PRIVATE cxx_std_17)
  set_target_properties(opencl-host-original PROPERTIES CXX_EXTENSIONS OFF)
  target_include_directories(opencl-host-original PRIVATE "${UDONCL_BINDINGS_DIR}")
  target_link_libraries(opencl-host-original PRIVATE VRCNativeOpenCLRuntime)
  if(UDONCL_BUILD_RUNTIME_TESTS)
    target_sources(opencl-host-original PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/runtime/opencl.c")
    target_compile_definitions(opencl-host-original PRIVATE UDONCL_BUILD_RUNTIME_TESTS=1)
  endif()
endif()
